Vec2: struct {
  x: i32;
  y: i32;
}

Line: struct {
  p1: Vec2;
  p2: Vec2;
}

extern print: (s: addr) -> i32;

core: mod {
  ops: mod {
    Add: interface {
      add: (self: Self, other: Self) -> Self;
    }
  }
}

Stringify: interface {
  to_str: (self: Self) -> str;
}

export main: () -> none {
  a := Vec2::new(34, 20);

  b := Vec2 {};
  b.x = 33;
  b.y = 400;

  res := a + b;

  print(res.to_str().c_str());

  line := Line::new(a, Vec2{ x=0, y=0 });
  line.p2.x = 6;
  line.p2.y = 7;

  print(Line::to_str(line).c_str());
}

impl core::ops::Add: str {
  add: (self: Self, other: Self) -> Self {
    return self.concat(other);
  }
}

impl Self: str {
  [[name="str__concat"]]
  extern concat: (self: Self, other: Self) -> Self;
  [[name="str__c_str"]]
  extern c_str: (self: Self) -> addr;
}

impl Stringify: i32 {
  [[name="i32__to_str"]]
  extern to_str: (self: Self) -> str;
}

impl core::ops::Add: Vec2 {
  add: (self: Self, other: Self) -> Self {
    return Vec2::new(
      self.x + other.x,
      self.y + other.y
    );
  }
}

impl Stringify: Vec2 {
  to_str: (self: Self) -> str {
    return "Vec2(" + self.x.to_str() + ", " + self.y.to_str() + ")";
  }
}

impl Self: Vec2 {
  new: (x: i32, y: i32) -> Self {
    return Self { x= x, y= y };
  }

  copy: (self: Self, other: Self) -> none {
    self.x = other.x;
    self.y = other.y;
  }
}

impl Stringify: Line {
  to_str: (self: Self) -> str {
    return "Line(" + self.p1.to_str() + ", " + self.p2.to_str() + ")";
  }
}

impl Self: Line {
  new: (p1: Vec2, p2: Vec2) -> Self {
    return Self { p1 = p1, p2 = p2 };
  }
}
