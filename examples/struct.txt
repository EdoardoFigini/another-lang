@import core;
@import std;

Vec2: struct {
  x: i32;
  y: i32;
}

Point: type Vec2;

Line: struct {
  p1: Point;
  p2: Point;
}

export main: () -> none {
  a := Vec2::make(34, 20);

  b := new Vec2{};
  b.x = 33;
  b.y = 400;

  res := a + b;

  std::io::println(res.to_str());

  line := Line::make(a, new Point{ x=0, y=0 });
  line.p2.x = 6;
  line.p2.y = 7;

  std::io::println(Line::to_str(line));
}

impl core::ops::Add: Vec2 {
  add: (self: Self, other: Self) -> Self {
    return Vec2::make(
      self.x + other.x,
      self.y + other.y
    );
  }
}

impl std::io::Stringify: Vec2 {
  to_str: (self: Self) -> str {
    return "Vec2(" + self.x.to_str() + ", " + self.y.to_str() + ")";
  }
}

impl Self: Vec2 {
  make: (x: i32, y: i32) -> Self {
    return new Self { x= x, y= y };
  }

  copy: (self: Self, other: Self) -> none {
    self.x = other.x;
    self.y = other.y;
  }
}

impl std::io::Stringify: Line {
  to_str: (self: Self) -> str {
    return "Line(" + self.p1.to_str() + ", " + self.p2.to_str() + ")";
  }
}

impl Self: Line {
  make: (p1: Vec2, p2: Vec2) -> Self {
    return new Self { p1 = p1, p2 = p2 };
  }
}
