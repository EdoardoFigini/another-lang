Node: struct {
  item: i32;
  next: Node;
}

List: struct {
  head: Node;
  tail: Node;
  _count: u32;
}

export main: () -> none {
  l := new List{};

  std::io::println(l.to_str() + " - count: " + (#l).to_str());

  i := 10;
  while i > 0 {
    l.insert(i);
    i = i - 1;
  }

  std::io::println(l.to_str() + " - count: " + (#l).to_str());

  i = 11;
  while i <= 20 {
    l.append(i);
    i = i + 1;
  }

  std::io::println(l.to_str() + " - count: " + (#l).to_str());
}

impl Self: List {
  make : () -> Self {
    return new Self { };
  }

  insert: (self: Self, n: i32) -> none {
    tmp := self.head;

    self.head = Node::make(n);
    self.head.next = tmp;

    if self._count == 0U {
      self.tail = self.head;
    }

    self._count = self._count + 1U;
  }

  append: (self: Self, n: i32) -> none {
    if self._count == 0U {
      self.insert(n);
      return;
    }

    self.tail.next = Node::make(n);
    self.tail = self.tail.next; 

    self._count = self._count + 1U;
  }

  to_str: (self: Self) -> str {
    if (self._count == 0U) { return "List()"; }
    s := "List(" + self.head.to_str();
    i := 1U;
    n := self.head.next;
    while (i < self._count) {
      s = s + ", " + n.to_str();

      n = n.next;
      i = i + 1U;
    }
    return s + ")";
  }
}

impl core::ops::Countable: List {
  count: (self: Self) -> u32 {
    return self._count;
  }
}

impl Self: Node {
  make : (n: i32) -> Self {
    return new Self { item = n };
  }

  to_str: (self: Self) -> str {
    return self.item.to_str();
  }
}
