N := 9U;

extern print: (s: addr) -> i32;

fib: (n : u32) -> u32 {
  if n < 2U {
    return n;
  } else {
    return fib(n - 1U) + fib(n - 2U);
  }
}

export main: () -> none {
  res := fib(N);
  print(res.to_str().c_str());
}

impl Self: u32 {
  [[name="i32__to_str"]]
  extern to_str: (self: Self) -> str;
}

impl Self: str {
  [[name="str__c_str"]]
  extern c_str: (self: Self) -> addr;
}
